async function makeUser(){const r=await shared.pool.query('INSERT INTO public.users(email) VALUES($1) RETURNING id',[`verify-sync-${randomUUID()}@example.test`]);shared.uid=r.rows[0].id;fixtures.push(shared.uid)}
async function push(word:string,base=0){
 const handler=(await import('__AUDIT_REPO_ROOT__/apps/server/src/api/core/progress.ts')).default
 const response=await handler(new Request('http://local.test/api/progress',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({learned:word?[word]:[],sync:{attemptId:randomUUID(),baseVersion:base,clientUpdatedAt:new Date().toISOString()}})}))
 return {status:response.status,body:await response.json()}
}
async function stored(){return (await shared.pool.query('SELECT learned,version FROM english.learning_progress WHERE user_id=$1',[shared.uid])).rows[0]}
test('serial first sync preserves union (negative control)',async()=>{
 await makeUser();shared.emptyBarrier=false
 expect((await push('cat')).status).toBe(200)
 expect((await push('dog',1)).status).toBe(200)
 expect((await stored()).learned.sort()).toEqual(['cat','dog'])
})
test('existing row concurrent updates preserve union (negative control)',async()=>{
 await makeUser();shared.emptyBarrier=false
 expect((await push('')).status).toBe(200)
 const results=await Promise.all([push('cat',1),push('dog',1)])
 expect(results.map(r=>r.status)).toEqual([200,200])
 const actual=await stored()
 expect(actual.learned.sort()).toEqual(['cat','dog'])
 expect(actual.version).toBe(3)
})
test('new row concurrent first sync must preserve both devices (regression)',async()=>{
